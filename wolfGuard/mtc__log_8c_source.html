67/******************************************************************************
68 * Function: mtc_log_init
69 *
70 * Description:
71 * Initialises the logging subsystem. Creates the parent directory if
72 * needed, opens the log file in append mode with line buffering, and
73 * records the initial log level. On any failure, falls back to stderr
74 * so that logging continues.
75 *
76 * Input Arguments:
77 * log_file - Path to the log file. NULL defaults to
78 * /var/log/mtc/mtc_server.log.
79 * level - Initial log level (MTC_LOG_ERROR..MTC_LOG_TRACE).
80 *
81 * Returns:
82 * 0 on success.
83 * -1 if the directory could not be created or the file could not be
84 * opened (logging falls back to stderr).
85 *
86 * Side Effects:
87 * - May call mkdir() to create the log directory.
88 * - Opens a file handle stored in s_logfp.
89 * - Writes an initial "logging started" message.
90 ******************************************************************************/
91int mtc_log_init(const char *log_file, int level)
92{
93 const char *path = log_file ? log_file : "/var/log/mtc/mtc_server.log";
94 char dir[512];
95 const char *last_slash;
96
97 s_level = level;
98
99 /* Extract parent directory and create if needed */
100 last_slash = strrchr(path, '/');
101 if (last_slash && (last_slash - path) < (int)sizeof(dir)) {
102 memcpy(dir, path, (size_t)(last_slash - path));
103 dir[last_slash - path] = '\0';
104 if (mkdir(dir, 0755) < 0 && errno != EEXIST) {
105 fprintf(stderr, "[log] cannot create %s: %s (logging to stderr)\n",
106 dir, strerror(errno));
107 s_logfp = stderr;
108 return -1;
109 }
110 }
111
112 s_logfp = fopen(path, "a");
113 if (!s_logfp) {
114 fprintf(stderr, "[log] cannot open %s: %s (logging to stderr)\n",
115 path, strerror(errno));
116 s_logfp = stderr;
117 return -1;
118 }
119
120 /* Line-buffered so messages appear promptly */
121 setvbuf(s_logfp, NULL, _IOLBF, 0);
122 s_to_file = 1;
123
124 mtc_log(MTC_LOG_INFO, "logging started (level=%d, file=%s)", level, path);
125 return 0;
126}

173/******************************************************************************
174 * Function: mtc_log
175 *
176 * Description:
177 * Core logging function. Formats a timestamped message with a level
178 * tag and writes it to both the log file (or stderr fallback) and
179 * stdout (for systemd journal capture). Messages above the current
180 * log level are silently dropped.
181 *
182 * Input Arguments:
183 * level - Message level. Dropped if > s_level.
184 * fmt - printf-style format string.
185 * ... - Format arguments.
186 *
187 * Side Effects:
188 * Writes to s_logfp and stdout.
189 ******************************************************************************/
190void mtc_log(int level, const char *fmt, ...)
191{
192 va_list ap;
193 time_t now;
194 struct tm tm;
195 char ts[32];
196
197 if (level > s_level)
198 return;
199
200 now = time(NULL);
201 localtime_r(&now, &tm);
202 strftime(ts, sizeof(ts), "%Y-%m-%d %H:%M:%S", &tm);
203
204 /* Write to log file (or stderr fallback) */
205 if (s_logfp) {
206 fprintf(s_logfp, "%s [%s] ", ts, level_str(level));
207 va_start(ap, fmt);
208 vfprintf(s_logfp, fmt, ap);
209 va_end(ap);
210 fprintf(s_logfp, "\n");
211 }
212
213 /* Also write to stdout for systemd journal */
214 printf("%s [%s] ", ts, level_str(level));
215 va_start(ap, fmt);
216 vprintf(fmt, ap);
217 va_end(ap);
218 printf("\n");
219}
