1/******************************************************************************
2 * File: mtc_ratelimit.c
3 * Purpose: Redis-backed per-IP rate limiting for the MTC CA server.
4 *
5 * Description:
6 * Implements sliding-window rate limiting using Redis INCR + EXPIRE
7 * counters. Each request increments two Redis keys per category
8 * (per-minute and per-hour), plus two more for the global per-IP
9 * limit. TTLs are set on the first increment so counters expire
10 * automatically.
11 *
12 * Key format:
13 * rl:<ip>:<category>:m — per-minute counter (TTL 60s)
14 * rl:<ip>:<category>:h — per-hour counter (TTL 3600s)
15 *
16 * If Redis is unavailable (connection failure or runtime error), all
17 * requests are allowed (fail-open policy).
18 *
19 * Dependencies:
20 * mtc_ratelimit.h
21 * mtc_log.h
22 * stdio.h, stdlib.h, string.h
23 * hiredis/hiredis.h (Redis C client)
24 *
25 * Notes:
26 * - NOT thread-safe. s_redis is file-scoped static.
27 * - Fail-open: Redis errors never block legitimate traffic.
28 * - Counters are always incremented, even when the request is denied,
29 * to prevent burst abuse after a limit reset.
30 *
31 * Created: 2026-04-13
32 ******************************************************************************/

63/******************************************************************************
64 * Function: mtc_ratelimit_init
65 *
66 * Description:
67 * Connects to Redis with a 1-second timeout. On failure, logs a
68 * warning and disables rate limiting (fail-open).
69 *
70 * Input Arguments:
71 * redis_host - Redis hostname. NULL defaults to "127.0.0.1".
72 * redis_port - Redis port. <= 0 defaults to 6379.
73 *
74 * Returns:
75 * 0 on success (Redis connected).
76 * -1 on connection failure (rate limiting disabled).
77 *
78 * Side Effects:
79 * Sets s_redis. Logs connection status.
80 ******************************************************************************/
81int mtc_ratelimit_init(const char *redis_host, int redis_port)
82{
83 struct timeval timeout = { 1, 0 }; /* 1-second connect timeout */
84
85 if (!redis_host) redis_host = "127.0.0.1";
86 if (redis_port <= 0) redis_port = 6379;
87
88 s_redis = redisConnectWithTimeout(redis_host, redis_port, timeout);
89 if (!s_redis || s_redis->err) {
90 LOG_WARN("Redis connect failed: %s — rate limiting disabled",
91 s_redis ? s_redis->errstr : "NULL");
92 if (s_redis) {
93 redisFree(s_redis);
94 s_redis = NULL;
95 }
96 return -1;
97 }
98
99 LOG_INFO("rate limiter connected to Redis %s:%d", redis_host, redis_port);
100 return 0;
101}

118/******************************************************************************
119 * Function: check_key
120 *
121 * Description:
122 * Atomically increments a Redis counter key and sets its TTL on the
123 * first increment (count == 1). This implements the sliding window:
124 * the key auto-expires after ttl_secs, resetting the counter.
125 *
126 * Input Arguments:
127 * key - Redis key string (e.g. "rl:1.2.3.4:0:m").
128 * ttl_secs - TTL in seconds (60 for per-minute, 3600 for per-hour).
129 *
130 * Returns:
131 * Current counter value after increment (>= 1).
132 * 0 on Redis error (fail-open — caller should allow the request).
133 ******************************************************************************/
134static int check_key(const char *key, int ttl_secs)
135{
136 redisReply *reply;
137 int count;
138
139 reply = redisCommand(s_redis, "INCR %s", key);
140 if (!reply || reply->type != REDIS_REPLY_INTEGER) {
141 if (reply) freeReplyObject(reply);
142 return 0; /* Redis error — fail-open */
143 }
144 count = (int)reply->integer;
145 freeReplyObject(reply);
146
147 /* Set TTL only on first increment so the window starts from the
148 * first request, not from some arbitrary point */
149 if (count == 1) {
150 reply = redisCommand(s_redis, "EXPIRE %s %d", key, ttl_secs);
151 if (reply) freeReplyObject(reply);
152 }
153
154 return count;
155}

157/******************************************************************************
158 * Function: mtc_ratelimit_check
159 *
160 * Description:
161 * Checks whether a request from the given IP is allowed under both
162 * the category-specific and global per-IP rate limits. Increments
163 * four Redis counters (category minute + hour, global minute + hour)
164 * and compares against the configured thresholds.
165 *
166 * If Redis is not connected, or ip is NULL/empty, or category is
167 * invalid, the request is always allowed (fail-open).
168 *
169 * Input Arguments:
170 * ip - Client IP address string.
171 * category - Rate limit category (RL_READ..RL_REVOKE).
172 *
173 * Returns:
174 * 1 if allowed.
175 * 0 if rate-limited (category or global limit exceeded).
176 *
177 * Side Effects:
178 * Increments Redis counters (even when denying the request).
179 * Logs rate limit hits at INFO level.
180 ******************************************************************************/
181int mtc_ratelimit_check(const char *ip, int category)
182{
183 char key_m[128], key_h[128];
184 char gkey_m[128], gkey_h[128];
185 int count_m, count_h, gcount_m, gcount_h;
186
187 if (!s_redis || !ip || !ip[0])
188 return 1; /* no Redis or no IP — fail-open */
189
190 if (category < 0 || category >= RL_NUM_CATEGORIES)
191 return 1;
192
193 /* Category-specific counters */
194 snprintf(key_m, sizeof(key_m), "rl:%s:%d:m", ip, category);
195 snprintf(key_h, sizeof(key_h), "rl:%s:%d:h", ip, category);
196
197 count_m = check_key(key_m, 60);
198 count_h = check_key(key_h, 3600);
199
200 if (count_m > s_limits[category].per_min) {
201 LOG_INFO("rate limit hit: %s %s %d/min (limit %d)",
202 ip, s_limits[category].name, count_m,
203 s_limits[category].per_min);
204 return 0;
205 }
206 if (count_h > s_limits[category].per_hour) {
207 LOG_INFO("rate limit hit: %s %s %d/hour (limit %d)",
208 ip, s_limits[category].name, count_h,
209 s_limits[category].per_hour);
210 return 0;
211 }
212
213 /* Global per-IP counters (always checked in addition to category) */
214 snprintf(gkey_m, sizeof(gkey_m), "rl:%s:%d:m", ip, RL_GLOBAL);
215 snprintf(gkey_h, sizeof(gkey_h), "rl:%s:%d:h", ip, RL_GLOBAL);
216
217 gcount_m = check_key(gkey_m, 60);
218 gcount_h = check_key(gkey_h, 3600);
219
220 if (gcount_m > s_limits[RL_GLOBAL].per_min) {
221 LOG_INFO("rate limit hit: %s global %d/min (limit %d)",
222 ip, gcount_m, s_limits[RL_GLOBAL].per_min);
223 return 0;
224 }
225 if (gcount_h > s_limits[RL_GLOBAL].per_hour) {
226 LOG_INFO("rate limit hit: %s global %d/hour (limit %d)",
227 ip, gcount_h, s_limits[RL_GLOBAL].per_hour);
228 return 0;
229 }
230
231 return 1; /* allowed */
232}
