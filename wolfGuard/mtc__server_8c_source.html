1/******************************************************************************
2 * File: mtc_server.c
3 * Purpose: MTC CA/Log Server entry point (C implementation).
4 *
5 * Description:
6 * Main entry point for the MTC (Merkle Tree Certificates) CA/Log server.
7 * Parses command-line arguments, initialises subsystems (logging, wolfSSL,
8 * rate limiter, AbuseIPDB, certificate store), and starts the blocking
9 * HTTP(-over-TLS) server.
10 *
11 * Equivalent to server/python/server.py but using wolfCrypt for crypto,
12 * json-c for JSON, and file-based storage with optional PostgreSQL (Neon)
13 * persistence.
14 *
15 * Build: make
16 * Usage: ./mtc_server [options] (run with -h for full option list)
17 *
18 * Dependencies:
19 * stdio.h, stdlib.h, string.h, signal.h
20 * wolfssl/options.h, wolfssl/ssl.h
21 * mtc_store.h (certificate store / Merkle tree)
22 * mtc_http.h (HTTP server)
23 * mtc_checkendpoint.h (AbuseIPDB)
24 * mtc_log.h (logging)
25 * mtc_ratelimit.h (Redis rate limiter)
26 *
27 * Notes:
28 * - Single-threaded. The server blocks in mtc_http_serve().
29 * - SIGPIPE is ignored so that closed-connection writes return errors
30 * rather than terminating the process.
31 * - Subsystem init failures (AbuseIPDB, Redis) are non-fatal; only
32 * mtc_store_init failure is fatal.
33 *
34 * Created: 2026-04-13
35 ******************************************************************************/
36

80/******************************************************************************
81 * Function: main
82 *
83 * Description:
84 * Server entry point. Parses command-line arguments and initialises
85 * subsystems in the following order:
86 *
87 * 1. Logging (mtc_log_init)
88 * 2. wolfSSL library (wolfSSL_Init)
89 * 3. SIGPIPE suppression
90 * 4. Redis rate limiter (non-fatal)
91 * 5. DB tokenpath (if --tokenpath)
92 * 6. AbuseIPDB module (non-fatal)
93 * 7. Certificate store (fatal on failure)
94 * 8. TLS configuration (optional)
95 * 9. HTTP server (blocks forever)
96 *
97 * On shutdown (unreachable in normal operation), cleans up the store,
98 * rate limiter, wolfSSL, and log file.
99 *
100 * Input Arguments:
101 * argc - Argument count.
102 * argv - Argument vector.
103 *
104 * Returns:
105 * 0 on clean exit.
106 * 1 if the MTC store failed to initialise.
107 ******************************************************************************/
108int main(int argc, char *argv[])
109{
110 setvbuf(stdout, NULL, _IONBF, 0);
111
112 const char *host = "0.0.0.0";
113 int port = 8443;
114 const char *data_dir = "./mtc-data";
115 const char *tokenpath = NULL;
116 const char *ca_name = "MTC-CA-C";
117 const char *log_id = "32473.2";
118 const char *tls_cert = NULL;
119 const char *tls_key = NULL;
120 const char *tls_ca = NULL;
121 const char *ech_name = NULL;
122 int log_level = MTC_LOG_INFO;
123 const char *log_file = NULL;
124 MtcStore store;
125 mtc_tls_cfg_t tls_cfg;
126 int i;
127
128 /* Parse command-line arguments */
129 for (i = 1; i < argc; i++) {
130 if (strcmp(argv[i], "--host") == 0 && i + 1 < argc)
131 host = argv[++i];
132 else if (strcmp(argv[i], "--port") == 0 && i + 1 < argc)
133 port = atoi(argv[++i]);
134 else if (strcmp(argv[i], "--data-dir") == 0 && i + 1 < argc)
135 data_dir = argv[++i];
136 else if (strcmp(argv[i], "--tokenpath") == 0 && i + 1 < argc)
137 tokenpath = argv[++i];
138 else if (strcmp(argv[i], "--ca-name") == 0 && i + 1 < argc)
139 ca_name = argv[++i];
140 else if (strcmp(argv[i], "--log-id") == 0 && i + 1 < argc)
141 log_id = argv[++i];
142 else if (strcmp(argv[i], "--abuse-threshold") == 0 && i + 1 < argc)
143 mtc_set_abuse_threshold(atoi(argv[++i]));
144 else if (strcmp(argv[i], "--tls-cert") == 0 && i + 1 < argc)
145 tls_cert = argv[++i];
146 else if (strcmp(argv[i], "--tls-key") == 0 && i + 1 < argc)
147 tls_key = argv[++i];
148 else if (strcmp(argv[i], "--tls-ca") == 0 && i + 1 < argc)
149 tls_ca = argv[++i];
150 else if (strcmp(argv[i], "--ech-name") == 0 && i + 1 < argc)
151 ech_name = argv[++i];
152 else if (strcmp(argv[i], "--log-level") == 0 && i + 1 < argc)
153 log_level = atoi(argv[++i]);
154 else if (strcmp(argv[i], "--log-file") == 0 && i + 1 < argc)
155 log_file = argv[++i];
156 else if (strcmp(argv[i], "-h") == 0) {
157 usage(argv[0]); return 0;
158 }
159 }
160
161 /* 1. Initialize logging */
162 mtc_log_init(log_file, log_level);
163
164 /* 2. Initialize wolfSSL library */
165 wolfSSL_Init();
166
167 /* 3. Ignore SIGPIPE — closed-connection writes return errors instead
168 * of killing the process */
169 signal(SIGPIPE, SIG_IGN);
170
171 /* 4. Initialize Redis-backed rate limiter (non-fatal if unavailable) */
172 mtc_ratelimit_init("127.0.0.1", 6379);
173
174 /* 5. Set token path for MERKLE_NEON DB connection string lookup */
175 if (tokenpath)
176 mtc_db_set_tokenpath(tokenpath);
177
178 /* 6. Initialize AbuseIPDB module (non-fatal if key missing) */
179 {
180 int rc = mtc_init();
181 if (rc == -2)
182 printf("[server] AbuseIPDB key not found, IP checking disabled\n");
183 else if (rc < 0)
184 fprintf(stderr, "[server] AbuseIPDB init failed (%d)\n", rc);
185 else
186 printf("[server] AbuseIPDB module initialized\n");
187 }
188
189 /* 7. Initialize certificate store — fatal on failure */
190 if (mtc_store_init(&store, data_dir, ca_name, log_id) != 0) {
191 fprintf(stderr, "Failed to initialize MTC store\n");
192 return 1;
193 }
194
195 /* 8. Set up TLS config (NULL if no --tls-cert → plain HTTP mode) */
196 memset(&tls_cfg, 0, sizeof(tls_cfg));
197 tls_cfg.cert_file = tls_cert;
198 tls_cfg.key_file = tls_key;
199 tls_cfg.ca_file = tls_ca;
200 tls_cfg.ech_public_name = ech_name;
201
202 /* 9. Run HTTP server (blocks indefinitely) */
203 mtc_http_serve(host, port, &store, tls_cert ? &tls_cfg : NULL);
204
205 /* Cleanup (unreachable in normal operation) */
206 mtc_store_free(&store);
207 mtc_ratelimit_close();
208 wolfSSL_Cleanup();
209 mtc_log_close();
210 return 0;
211}
