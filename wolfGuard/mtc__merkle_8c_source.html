1/******************************************************************************
2 * File: mtc_merkle.c
3 * Purpose: Merkle tree implementation for the MTC CA server.
4 *
5 * Description:
6 * Implements an append-only Merkle hash tree following RFC 9162
7 * Section 2.1 (Certificate Transparency v2) conventions:
8 *
9 * Leaf hash: SHA-256(0x00 || data)
10 * Node hash: SHA-256(0x01 || left || right)
11 *
12 * Tree hashes are computed on demand (no internal cache). Inclusion
13 * proofs follow RFC 9162 Section 2.1.3 (PATH) and consistency proofs
14 * follow Section 2.1.4 (SUBPROOF).
15 *
16 * Dependencies:
17 * mtc_merkle.h
18 * stdlib.h, string.h
19 * wolfssl/options.h
20 * wolfssl/wolfcrypt/sha256.h
21 *
22 * Notes:
23 * - NOT thread-safe. All tree operations must be serialised.
24 * - The tree owns all memory in its arrays. mtc_tree_free() releases
25 * everything.
26 * - Proof buffers returned by inclusion/consistency functions are
27 * malloc'd; the caller must free().
28 *
29 * Created: 2026-04-13
30 ******************************************************************************/

38/******************************************************************************
39 * Function: mtc_hash_leaf
40 *
41 * Description:
42 * Computes the leaf hash per RFC 9162 Section 2.1:
43 * SHA-256(0x00 || data). The 0x00 prefix distinguishes leaf hashes
44 * from interior node hashes, preventing second-preimage attacks.
45 *
46 * Input Arguments:
47 * data - Leaf data bytes.
48 * dataSz - Size of data in bytes.
49 * out - Caller-owned buffer for the 32-byte hash.
50 ******************************************************************************/
51void mtc_hash_leaf(const uint8_t *data, int dataSz, uint8_t *out)
52{
53 wc_Sha256 sha;
54 uint8_t prefix = 0x00; /* RFC 9162 leaf domain separator */
55 wc_InitSha256(&sha);
56 wc_Sha256Update(&sha, &prefix, 1);
57 wc_Sha256Update(&sha, data, (word32)dataSz);
58 wc_Sha256Final(&sha, out);
59 wc_Sha256Free(&sha);
60}

62/******************************************************************************
63 * Function: mtc_hash_node
64 *
65 * Description:
66 * Computes an interior node hash per RFC 9162 Section 2.1:
67 * SHA-256(0x01 || left || right). The 0x01 prefix distinguishes node
68 * hashes from leaf hashes.
69 *
70 * Input Arguments:
71 * left - Left child hash (MTC_HASH_SIZE bytes).
72 * right - Right child hash (MTC_HASH_SIZE bytes).
73 * out - Caller-owned buffer for the 32-byte hash.
74 ******************************************************************************/
75void mtc_hash_node(const uint8_t *left, const uint8_t *right, uint8_t *out)
76{
77 wc_Sha256 sha;
78 uint8_t prefix = 0x01; /* RFC 9162 node domain separator */
79 wc_InitSha256(&sha);
80 wc_Sha256Update(&sha, &prefix, 1);
81 wc_Sha256Update(&sha, left, MTC_HASH_SIZE);
82 wc_Sha256Update(&sha, right, MTC_HASH_SIZE);
83 wc_Sha256Final(&sha, out);
84 wc_Sha256Free(&sha);
85}

87/******************************************************************************
88 * Function: mtc_tree_init
89 *
90 * Description:
91 * Initialises an empty Merkle tree with an initial capacity of 64 slots.
92 *
93 * Input Arguments:
94 * tree - Tree struct to initialise. Zeroed and allocated on return.
95 *
96 * Side Effects:
97 * Allocates three arrays (leaf_hashes, entries, entry_sizes).
98 ******************************************************************************/
99void mtc_tree_init(MtcMerkleTree *tree)
100{
101 memset(tree, 0, sizeof(*tree));
102 tree->capacity = 64;
103 tree->leaf_hashes = (uint8_t**)calloc((size_t)tree->capacity, sizeof(uint8_t*));
104 tree->entries = (uint8_t**)calloc((size_t)tree->capacity, sizeof(uint8_t*));
105 tree->entry_sizes = (int*)calloc((size_t)tree->capacity, sizeof(int));
106}

108/******************************************************************************
109 * Function: mtc_tree_free
110 *
111 * Description:
112 * Frees all memory owned by the tree: each leaf hash, each entry buffer,
113 * the three top-level arrays, and zeros the struct. Safe to re-init
114 * after this call.
115 *
116 * Input Arguments:
117 * tree - Tree to free.
118 ******************************************************************************/
119void mtc_tree_free(MtcMerkleTree *tree)
120{
121 int i;
122 for (i = 0; i < tree->size; i++) {
123 free(tree->leaf_hashes[i]);
124 free(tree->entries[i]);
125 }
126 free(tree->leaf_hashes);
127 free(tree->entries);
128 free(tree->entry_sizes);
129 memset(tree, 0, sizeof(*tree));
130}

132/******************************************************************************
133 * Function: mtc_tree_grow
134 *
135 * Description:
136 * Doubles the capacity of the tree's internal arrays when full.
137 *
138 * Input Arguments:
139 * tree - Tree to grow (only reallocates if size >= capacity).
140 ******************************************************************************/
141static void mtc_tree_grow(MtcMerkleTree *tree)
142{
143 if (tree->size >= tree->capacity) {
144 int newcap = tree->capacity * 2;
145 tree->leaf_hashes = (uint8_t**)realloc(tree->leaf_hashes,
146 (size_t)newcap * sizeof(uint8_t*));
147 tree->entries = (uint8_t**)realloc(tree->entries,
148 (size_t)newcap * sizeof(uint8_t*));
149 tree->entry_sizes = (int*)realloc(tree->entry_sizes,
150 (size_t)newcap * sizeof(int));
151 tree->capacity = newcap;
152 }
153}

155/******************************************************************************
156 * Function: mtc_tree_append
157 *
158 * Description:
159 * Appends an entry to the tree. Copies the entry data, computes the
160 * leaf hash, and stores both. Grows arrays if at capacity.
161 *
162 * Input Arguments:
163 * tree - Target tree.
164 * entry - Serialised entry bytes (copied; caller retains ownership).
165 * entrySz - Size of entry in bytes.
166 *
167 * Returns:
168 * 0-based index of the newly appended entry.
169 *
170 * Side Effects:
171 * Allocates MTC_HASH_SIZE bytes for the leaf hash and entrySz bytes
172 * for the entry copy. Both are owned by the tree.
173 ******************************************************************************/
174int mtc_tree_append(MtcMerkleTree *tree, const uint8_t *entry, int entrySz)
175{
176 int idx = tree->size;
177 uint8_t *lh, *ent;
178
179 mtc_tree_grow(tree);
180
181 lh = (uint8_t*)malloc(MTC_HASH_SIZE);
182 mtc_hash_leaf(entry, entrySz, lh);
183
184 ent = (uint8_t*)malloc((size_t)entrySz);
185 memcpy(ent, entry, (size_t)entrySz);
186
187 tree->leaf_hashes[idx] = lh;
188 tree->entries[idx] = ent;
189 tree->entry_sizes[idx] = entrySz;
190 tree->size++;
191
192 return idx;
193}

195/******************************************************************************
196 * Function: mth
197 *
198 * Description:
199 * Computes MTH(D[start:end]) — the Merkle Tree Hash over a contiguous
200 * range of leaves, per RFC 9162 Section 2.1. Recurses by splitting
201 * at the largest power of 2 less than (end - start).
202 *
203 * Input Arguments:
204 * tree - Merkle tree.
205 * start - First leaf index (inclusive).
206 * end - One past the last leaf (exclusive). Must satisfy end > start.
207 * out - Caller-owned buffer for the MTC_HASH_SIZE-byte result.
208 *
209 * Notes:
210 * For a single leaf (n == 1), returns the precomputed leaf hash directly.
211 ******************************************************************************/
212static void mth(MtcMerkleTree *tree, int start, int end, uint8_t *out)
213{
214 int n = end - start;
215 int k;
216 uint8_t left[MTC_HASH_SIZE], right[MTC_HASH_SIZE];
217
218 if (n == 1) {
219 memcpy(out, tree->leaf_hashes[start], MTC_HASH_SIZE);
220 return;
221 }
222
223 /* k = largest power of 2 less than n (RFC 9162 split point) */
224 k = 1;
225 while (k * 2 < n) k *= 2;
226
227 mth(tree, start, start + k, left);
228 mth(tree, start + k, end, right);
229 mtc_hash_node(left, right, out);
230}

232/******************************************************************************
233 * Function: mtc_tree_root_hash
234 *
235 * Description:
236 * Computes the root hash of the first tree_size leaves. If tree_size
237 * is invalid (<= 0 or > tree->size), returns SHA-256("") — the
238 * conventional empty-tree hash.
239 *
240 * Input Arguments:
241 * tree - Merkle tree.
242 * tree_size - Number of leaves to include.
243 * out - Caller-owned buffer for the MTC_HASH_SIZE-byte hash.
244 *
245 * Returns:
246 * 0 always.
247 ******************************************************************************/
248int mtc_tree_root_hash(MtcMerkleTree *tree, int tree_size, uint8_t *out)
249{
250 if (tree_size <= 0 || tree_size > tree->size) {
251 /* Empty tree: SHA-256 of the empty string */
252 wc_Sha256 sha;
253 wc_InitSha256(&sha);
254 wc_Sha256Final(&sha, out);
255 wc_Sha256Free(&sha);
256 return 0;
257 }
258 mth(tree, 0, tree_size, out);
259 return 0;
260}

286/******************************************************************************
287 * Function: inclusion_path
288 *
289 * Description:
290 * Recursive helper implementing PATH(m, D_n) per RFC 9162 Section 2.1.3.
291 * Builds the inclusion proof by appending sibling hashes to the proof
292 * buffer as it recurses.
293 *
294 * Input Arguments:
295 * tree - Merkle tree.
296 * m - Leaf index relative to the current subtree (0-based).
297 * n - Size of the current subtree.
298 * offset - Absolute offset of the current subtree within the tree.
299 * proof - Pre-allocated output buffer for sibling hashes.
300 * count - Pointer to the current proof element count (updated in place).
301 *
302 * Notes:
303 * Base case: n == 1 (single leaf) — no siblings to add.
304 * Split point k is the largest power of 2 less than n.
305 ******************************************************************************/
306static void inclusion_path(MtcMerkleTree *tree, int m, int n, int offset,
307 uint8_t *proof, int *count)
308{
309 int k;
310 uint8_t h[MTC_HASH_SIZE];
311
312 if (n == 1)
313 return;
314
315 /* RFC 9162 split point: largest power of 2 less than n */
316 k = 1;
317 while (k * 2 < n) k *= 2;
318
319 if (m < k) {
320 /* Target is in the left subtree — recurse left, hash right sibling */
321 inclusion_path(tree, m, k, offset, proof, count);
322 mth(tree, offset + k, offset + n, h);
323 }
324 else {
325 /* Target is in the right subtree — recurse right, hash left sibling */
326 inclusion_path(tree, m - k, n - k, offset + k, proof, count);
327 mth(tree, offset, offset + k, h);
328 }
329 memcpy(proof + (*count) * MTC_HASH_SIZE, h, MTC_HASH_SIZE);
330 (*count)++;
331}

333/******************************************************************************
334 * Function: mtc_tree_inclusion_proof
335 *
336 * Description:
337 * Generates a Merkle inclusion proof for leaf 'index' within the
338 * subtree [start, end). Allocates a proof buffer and returns it.
339 *
340 * Input Arguments:
341 * tree - Merkle tree.
342 * index - Leaf index to prove (absolute, within [start, end)).
343 * start - Subtree start (inclusive).
344 * end - Subtree end (exclusive).
345 * proof_out - Receives a malloc'd array of sibling hashes.
346 * proof_count - Receives the number of hashes in the proof.
347 *
348 * Returns:
349 * 0 on success.
350 * -1 if index is out of [start, end) or end > tree->size.
351 *
352 * Notes:
353 * Caller must free(*proof_out).
354 ******************************************************************************/
355int mtc_tree_inclusion_proof(MtcMerkleTree *tree, int index, int start, int end,
356 uint8_t **proof_out, int *proof_count)
357{
358 int max_depth, count = 0;
359 uint8_t *proof;
360
361 if (index < start || index >= end || end > tree->size)
362 return -1;
363
364 /* Conservative upper bound on proof depth: ceil(log2(n)) + 2 */
365 max_depth = 0;
366 { int tmp = end - start; while (tmp > 1) { max_depth++; tmp /= 2; } max_depth += 2; }
367
368 proof = (uint8_t*)malloc((size_t)max_depth * MTC_HASH_SIZE);
369
370 inclusion_path(tree, index - start, end - start, start, proof, &count);
371
372 *proof_out = proof;
373 *proof_count = count;
374 return 0;
375}

377/******************************************************************************
378 * Function: consistency_subproof
379 *
380 * Description:
381 * Recursive helper implementing SUBPROOF(m, D_n) per RFC 9162
382 * Section 2.1.4. Builds the consistency proof by appending hashes
383 * to the proof buffer.
384 *
385 * Input Arguments:
386 * tree - Merkle tree.
387 * m - Old tree size relative to the current subtree.
388 * n - Current subtree size.
389 * offset - Absolute offset of the current subtree.
390 * start_from_old - 1 if this is the initial call (m == old_size),
391 * 0 for recursive calls. When 1 and m == n, the
392 * subtree hash is omitted (RFC 9162 optimisation).
393 * proof - Pre-allocated output buffer for hashes.
394 * count - Pointer to the current proof element count.
395 *
396 * Notes:
397 * Base case: m == n. If start_from_old is true, no hash is emitted
398 * (the verifier already knows the old root).
399 ******************************************************************************/
400static void consistency_subproof(MtcMerkleTree *tree, int m, int n, int offset,
401 int start_from_old, uint8_t *proof, int *count)
402{
403 int k;
404 uint8_t h[MTC_HASH_SIZE];
405
406 if (m == n) {
407 if (!start_from_old) {
408 /* Emit the subtree hash for the verifier */
409 mth(tree, offset, offset + n, h);
410 memcpy(proof + (*count) * MTC_HASH_SIZE, h, MTC_HASH_SIZE);
411 (*count)++;
412 }
413 return;
414 }
415
416 /* RFC 9162 split point: largest power of 2 less than n */
417 k = 1;
418 while (k * 2 < n) k *= 2;
419
420 if (m <= k) {
421 /* Old tree fits entirely in the left subtree */
422 consistency_subproof(tree, m, k, offset, start_from_old, proof, count);
423 mth(tree, offset + k, offset + n, h);
424 }
425 else {
426 /* Old tree straddles the split — recurse into the right half */
427 consistency_subproof(tree, m - k, n - k, offset + k, 0, proof, count);
428 mth(tree, offset, offset + k, h);
429 }
430 memcpy(proof + (*count) * MTC_HASH_SIZE, h, MTC_HASH_SIZE);
431 (*count)++;
432}

435 * Function: mtc_tree_consistency_proof
436 *
437 * Description:
438 * Generates a Merkle consistency proof from old_size to new_size,
439 * per RFC 9162 Section 2.1.4. Allocates a proof buffer and returns it.
440 *
441 * Input Arguments:
442 * tree - Merkle tree.
443 * old_size - Smaller tree size (>= 1).
444 * new_size - Larger tree size (<= tree->size).
445 * proof_out - Receives a malloc'd array of hashes.
446 * proof_count - Receives the number of hashes in the proof.
447 *
448 * Returns:
449 * 0 on success.
450 * -1 if old_size < 1, new_size > tree->size, or old_size > new_size.
451 *
452 * Notes:
453 * Caller must free(*proof_out).
454 ******************************************************************************/
455int mtc_tree_consistency_proof(MtcMerkleTree *tree, int old_size, int new_size,
456 uint8_t **proof_out, int *proof_count)
457{
458 int max_depth, count = 0;
459 uint8_t *proof;
460
461 if (old_size < 1 || new_size > tree->size || old_size > new_size)
462 return -1;
463
464 /* Conservative upper bound on proof depth: ceil(log2(n)) + 2 */
465 max_depth = 0;
466 { int tmp = new_size; while (tmp > 1) { max_depth++; tmp /= 2; } max_depth += 2; }
467
468 proof = (uint8_t*)malloc((size_t)max_depth * MTC_HASH_SIZE);
469 consistency_subproof(tree, old_size, new_size, 0, 1, proof, &count);
470
471 *proof_out = proof;
472 *proof_count = count;
473 return 0;
