1/******************************************************************************
2 * File: mtc_store.c
3 * Purpose: Persistence and CA operations for the MTC CA server.
4 *
5 * Description:
6 * Manages all server-side state: the Merkle tree, Ed25519 CA key,
7 * certificates, checkpoints, landmarks, and revocations. Supports two
8 * storage backends:
9 * - PostgreSQL (Neon) when MERKLE_NEON is available
10 * - File-based JSON (entries.json, certificates.json, landmarks.json,
11 * revocations.json) in data_dir as fallback
12 *
13 * The CA Ed25519 key is loaded from DB, file (ca_key.der), or
14 * generated fresh on first run.
15 *
16 * Dependencies:
17 * mtc_store.h, mtc_log.h
18 * stdio.h, stdlib.h, string.h, time.h
19 * sys/stat.h (mkdir, chmod)
20 * wolfssl/options.h
21 * wolfssl/wolfcrypt/ed25519.h (CA key operations)
22 * wolfssl/wolfcrypt/random.h (key generation)
23 * wolfssl/wolfcrypt/asn_public.h (DER/PEM conversion)
24 *
25 * Notes:
26 * - NOT thread-safe. All operations must be serialised.
27 * - The store owns all json_object refs in certificates/checkpoints.
28 * - CA private key is stored as DER in ca_key.der (chmod 0600) and
29 * optionally mirrored to DB as hex (unencrypted — see warning in
30 * init_ca_key).
31 *
32 * Created: 2026-04-13
33 ******************************************************************************/

52/******************************************************************************
53 * Function: mkdirp
54 *
55 * Description:
56 * Recursively creates directories for the given path, similar to
57 * "mkdir -p". Walks the path string, creating each component.
58 *
59 * Input Arguments:
60 * path - Directory path to create (may already exist).
61 ******************************************************************************/
62static void mkdirp(const char *path)
63{
64 char tmp[512];
65 char *p;
66 snprintf(tmp, sizeof(tmp), "%s", path);
67 for (p = tmp + 1; *p; p++) {
68 if (*p == '/') {
69 *p = '\0';
70 mkdir(tmp, 0700);
71 *p = '/';
72 }
73 }
74 mkdir(tmp, 0700);
75}

77/******************************************************************************
78 * Function: write_file
79 *
80 * Description:
81 * Writes raw bytes to a file, overwriting any existing content.
82 *
83 * Input Arguments:
84 * path - File path.
85 * data - Data to write.
86 * sz - Number of bytes to write.
87 *
88 * Returns:
89 * 0 on success.
90 * -1 if the file could not be opened.
91 ******************************************************************************/
92static int write_file(const char *path, const void *data, int sz)
93{
94 FILE *f = fopen(path, "wb");
95 if (!f) return -1;
96 fwrite(data, 1, (size_t)sz, f);
97 fclose(f);
98 return 0;
99}

101/******************************************************************************
102 * Function: read_file
103 *
104 * Description:
105 * Reads an entire file into a caller-owned buffer.
106 *
107 * Input Arguments:
108 * path - File path.
109 * buf - Caller-owned destination buffer.
110 * maxSz - Maximum bytes to read (rejects files larger than this).
111 *
112 * Returns:
113 * Number of bytes read (>= 0) on success.
114 * -1 if the file could not be opened or exceeds maxSz.
115 ******************************************************************************/
116static int read_file(const char *path, void *buf, int maxSz)
117{
118 FILE *f = fopen(path, "rb");
119 long sz;
120 if (!f) return -1;
121 fseek(f, 0, SEEK_END);
122 sz = ftell(f);
123 fseek(f, 0, SEEK_SET);
124 if (sz > maxSz) { fclose(f); return -1; }
125 sz = (long)fread(buf, 1, (size_t)sz, f);
126 fclose(f);
127 return (int)sz;
128}

134/******************************************************************************
135 * Function: init_ca_key
136 *
137 * Description:
138 * Loads or generates the CA Ed25519 signing key. Sources are tried in
139 * order:
140 * 1. Database (ca_private_key_hex in mtc_ca_config)
141 * 2. File (data_dir/ca_key.der)
142 * 3. Generate new key, save to file (chmod 0600) and DB
143 *
144 * Input Arguments:
145 * store - Store with data_dir and DB connection already set.
146 *
147 * Returns:
148 * 0 on success, non-zero wolfCrypt error code on failure.
149 *
150 * Side Effects:
151 * Populates store->ca_priv_key, ca_priv_key_sz, ca_pub_key, ca_pub_key_sz.
152 * May write ca_key.der and/or DB config row.
153 ******************************************************************************/
154static int init_ca_key(MtcStore *store)
155{
156 char path[1024];
157 ed25519_key key;
158 WC_RNG rng;
159 int ret;
160
161 /* Try to load from DB first */
162 if (store->use_db && store->db) {
163 char *hex = mtc_db_load_config(store->db, "ca_private_key_hex");
164 if (hex) {
165 int i, len = (int)strlen(hex) / 2;
166 if (len <= (int)sizeof(store->ca_priv_key)) {
167 for (i = 0; i < len; i++) {
168 unsigned int bv;
169 sscanf(hex + i * 2, "%02x", &bv);
170 store->ca_priv_key[i] = (uint8_t)bv;
171 }
172 store->ca_priv_key_sz = len;
173 LOG_WARN("CA private key loaded from database — key is "
174 "stored unencrypted in PostgreSQL. Consider using "
175 "file-based storage (ca_key.der) for production.");
176 }
177 free(hex);
178 }
179 }
180
181 snprintf(path, sizeof(path), "%s/ca_key.der", store->data_dir);
182
183 /* Try to load from file if not from DB */
184 if (store->ca_priv_key_sz <= 0)
185 store->ca_priv_key_sz = read_file(path, store->ca_priv_key,
186 (int)sizeof(store->ca_priv_key));
187
188 if (store->ca_priv_key_sz > 0) {
189 /* Extract public key from private */
190 word32 idx = 0;
191 ret = wc_ed25519_init(&key);
192 if (ret != 0) return ret;
193 ret = wc_Ed25519PrivateKeyDecode(store->ca_priv_key, &idx, &key,
194 (word32)store->ca_priv_key_sz);
195 if (ret == 0) {
196 word32 pubSz = sizeof(store->ca_pub_key);
197 ret = wc_ed25519_export_public(&key, store->ca_pub_key, &pubSz);
198 store->ca_pub_key_sz = (int)pubSz;
199 }
200 wc_ed25519_free(&key);
201 return ret;
202 }
203
204 /* Generate new Ed25519 key */
205 printf("[store] generating Ed25519 CA key ...\n");
206 ret = wc_InitRng(&rng);
207 if (ret != 0) return ret;
208
209 ret = wc_ed25519_init(&key);
210 if (ret == 0)
211 ret = wc_ed25519_make_key(&rng, ED25519_KEY_SIZE, &key);
212 wc_FreeRng(&rng);
213 if (ret != 0) { wc_ed25519_free(&key); return ret; }
214
215 /* Export private key DER */
216 {
217 word32 privSz = sizeof(store->ca_priv_key);
218 ret = wc_Ed25519KeyToDer(&key, store->ca_priv_key, privSz);
219 if (ret > 0) {
220 store->ca_priv_key_sz = ret;
221 write_file(path, store->ca_priv_key, store->ca_priv_key_sz);
222 chmod(path, 0600);
223
224 /* Also save to DB */
225 if (store->use_db && store->db) {
226 char hex[256];
227 int j;
228 for (j = 0; j < store->ca_priv_key_sz; j++)
229 snprintf(hex + j * 2, 3, "%02x", store->ca_priv_key[j]);
230 mtc_db_save_config(store->db, "ca_private_key_hex", hex);
231 }
232 ret = 0;
233 }
234 }
235
236 /* Export public key */
237 {
238 word32 pubSz = sizeof(store->ca_pub_key);
239 wc_ed25519_export_public(&key, store->ca_pub_key, &pubSz);
240 store->ca_pub_key_sz = (int)pubSz;
241 }
242
243 wc_ed25519_free(&key);
244 return ret;
245}

251/******************************************************************************
252 * Function: mtc_store_init
253 *
254 * Description:
255 * Initialises the store: creates the data directory, connects to
256 * PostgreSQL (if available), initialises the Merkle tree, allocates
257 * certificate/checkpoint arrays, loads or generates the CA key, and
258 * restores persisted state. If the tree is empty after loading, a
259 * null entry (0x00) is inserted at index 0.
260 *
261 * Input Arguments:
262 * store - Store to initialise.
263 * data_dir - Directory for file-based storage.
264 * ca_name - CA display name.
265 * log_id - Log identifier string.
266 *
267 * Returns:
268 * 0 on success.
269 * -1 if the CA key could not be initialised.
270 *
271 * Side Effects:
272 * Creates data_dir, connects to DB, allocates arrays, may generate
273 * and persist a new CA key.
274 ******************************************************************************/
275int mtc_store_init(MtcStore *store, const char *data_dir,
276 const char *ca_name, const char *log_id)
277{
278 memset(store, 0, sizeof(*store));
279 snprintf(store->data_dir, sizeof(store->data_dir), "%s", data_dir);
280 snprintf(store->ca_name, sizeof(store->ca_name), "%s", ca_name);
281 snprintf(store->log_id, sizeof(store->log_id), "%s", log_id);
282 snprintf(store->cosigner_id, sizeof(store->cosigner_id), "%s.ca", log_id);
283
284 mkdirp(data_dir);
285
286 /* Try to connect to PostgreSQL (Neon) */
287 if (mtc_db_get_connstr() != NULL) {
288 store->db = mtc_db_connect();
289 if (store->db) {
290 store->use_db = 1;
291 mtc_db_init_schema(store->db);
292 printf("[store] using PostgreSQL (Neon) for persistence\n");
293 fflush(stdout);
294 }
295 else {
296 printf("[store] PostgreSQL unavailable, falling back to files\n");
297 fflush(stdout);
298 }
299 }
300 else {
301 printf("[store] MERKLE_NEON not set, using file-based storage\n");
302 fflush(stdout);
303 }
304
305 mtc_tree_init(&store->tree);
306
307 store->cert_capacity = 256;
308 store->certificates = (struct json_object**)calloc(
309 (size_t)store->cert_capacity, sizeof(struct json_object*));
310
311 store->checkpoints = (struct json_object**)calloc(256,
312 sizeof(struct json_object*));
313
314 /* Load or generate CA key */
315 if (init_ca_key(store) != 0) {
316 fprintf(stderr, "failed to initialize CA key\n");
317 return -1;
318 }
319
320 /* Try to load existing state */
321 mtc_store_load(store);
322
323 /* If empty, add null entry (index 0) */
324 if (store->tree.size == 0) {
325 uint8_t null_entry = 0x00;
326 mtc_tree_append(&store->tree, &null_entry, 1);
327 }
328
329 return 0;
330}

332/******************************************************************************
333 * Function: mtc_store_free
334 *
335 * Description:
336 * Frees all memory owned by the store: the Merkle tree, all certificate
337 * and checkpoint json_objects, and the top-level arrays. Does NOT close
338 * the DB connection.
339 *
340 * Input Arguments:
341 * store - Store to free.
342 ******************************************************************************/
343void mtc_store_free(MtcStore *store)
344{
345 int i;
346 mtc_tree_free(&store->tree);
347 for (i = 0; i < store->cert_count; i++) {
348 if (store->certificates[i])
349 json_object_put(store->certificates[i]);
350 }
351 free(store->certificates);
352 for (i = 0; i < store->checkpoint_count; i++) {
353 if (store->checkpoints[i])
354 json_object_put(store->checkpoints[i]);
355 }
356 free(store->checkpoints);
357}

363/******************************************************************************
364 * Function: mtc_store_save
365 *
366 * Description:
367 * Persists the current store state to JSON files in data_dir:
368 * entries.json — hex-encoded tree entries
369 * certificates.json — issued certificate objects
370 * landmarks.json — landmark tree sizes
371 *
372 * Input Arguments:
373 * store - Store to save.
374 *
375 * Returns:
376 * 0 always.
377 ******************************************************************************/
378int mtc_store_save(MtcStore *store)
379{
380 char path[1024];
381 int i;
382
383 /* Save entries as a JSON array */
384 {
385 struct json_object *arr = json_object_new_array();
386 for (i = 0; i < store->tree.size; i++) {
387 struct json_object *entry = json_object_new_object();
388 int esz = store->tree.entry_sizes[i];
389 int hexsz = esz * 2 + 1;
390 char *hex = (char*)malloc(hexsz);
391 int j;
392 for (j = 0; j < esz; j++)
393 snprintf(hex + j * 2, 3, "%02x", store->tree.entries[i][j]);
394 json_object_object_add(entry, "hex", json_object_new_string(hex));
395 json_object_object_add(entry, "size",
396 json_object_new_int(esz));
397 json_object_array_add(arr, entry);
398 free(hex);
399 }
400 snprintf(path, sizeof(path), "%s/entries.json", store->data_dir);
401 {
402 const char *s = json_object_to_json_string_ext(arr,
403 JSON_C_TO_STRING_PRETTY);
404 write_file(path, s, (int)strlen(s));
405 }
406 json_object_put(arr);
407 }
408
409 /* Save certificates */
410 {
411 struct json_object *arr = json_object_new_array();
412 for (i = 0; i < store->cert_count; i++) {
413 if (store->certificates[i])
414 json_object_array_add(arr, json_object_get(store->certificates[i]));
415 }
416 snprintf(path, sizeof(path), "%s/certificates.json", store->data_dir);
417 {
418 const char *s = json_object_to_json_string_ext(arr,
419 JSON_C_TO_STRING_PRETTY);
420 write_file(path, s, (int)strlen(s));
421 }
422 json_object_put(arr);
423 }
424
425 /* Save landmarks */
426 {
427 struct json_object *arr = json_object_new_array();
428 for (i = 0; i < store->landmark_count; i++)
429 json_object_array_add(arr,
430 json_object_new_int(store->landmarks[i]));
431 snprintf(path, sizeof(path), "%s/landmarks.json", store->data_dir);
432 {
433 const char *s = json_object_to_json_string(arr);
434 write_file(path, s, (int)strlen(s));
435 }
436 json_object_put(arr);
437 }
438
439 return 0;
440}

442/******************************************************************************
443 * Function: mtc_store_load
444 *
445 * Description:
446 * Loads persisted state into the store. If use_db is set, loads from
447 * PostgreSQL (entries, landmarks, certificates, revocations). Otherwise
448 * loads from JSON files in data_dir (entries.json, certificates.json,
449 * landmarks.json).
450 *
451 * Input Arguments:
452 * store - Store to populate (tree and arrays are appended to).
453 *
454 * Returns:
455 * 0 always (an empty result is not an error).
456 *
457 * Side Effects:
458 * Appends entries to store->tree, populates certificates, landmarks,
459 * and revocations arrays.
460 ******************************************************************************/
461int mtc_store_load(MtcStore *store)
462{
463 char path[1024], buf[1024 * 1024];
464 int sz;
465 struct json_object *arr, *entry;
466 int i, count;
467
468 /* Load from PostgreSQL if available */
469 if (store->use_db && store->db) {
470 struct json_object *db_entries = NULL;
471 int n;
472
473 /* Entries */
474 n = mtc_db_load_entries(store->db, &db_entries);
475 if (n > 0 && db_entries) {
476 count = (int)json_object_array_length(db_entries);
477 for (i = 0; i < count; i++) {
478 struct json_object *e = json_object_array_get_idx(db_entries, (size_t)i);
479 struct json_object *val;
480 if (json_object_object_get_ex(e, "serialized_hex", &val)) {
481 const char *hex = json_object_get_string(val);
482 int entry_sz = 0;
483 int j;
484
485 if (json_object_object_get_ex(e, "serialized_len", &val))
486 entry_sz = json_object_get_int(val);
487
488 if (entry_sz > 0) {
489 uint8_t *entry_bytes = (uint8_t*)malloc(entry_sz);
490 for (j = 0; j < entry_sz && hex[j*2] && hex[j*2+1]; j++) {
491 unsigned int bv;
492 sscanf(hex + j * 2, "%02x", &bv);
493 entry_bytes[j] = (uint8_t)bv;
494 }
495 mtc_tree_append(&store->tree, entry_bytes, entry_sz);
496 free(entry_bytes);
497 }
498 }
499 }
500 json_object_put(db_entries);
501 }
502
503 /* Landmarks */
504 store->landmark_count = mtc_db_load_landmarks(store->db,
505 store->landmarks, MTC_MAX_LANDMARKS);
506
507 /* Certificates */
508 {
509 struct json_object **certs = NULL;
510 int cert_count = 0;
511 mtc_db_load_all_certificates(store->db, &certs, &cert_count);
512 if (certs && cert_count > 0) {
513 if (cert_count > store->cert_capacity) {
514 store->cert_capacity = cert_count * 2;
515 store->certificates = (struct json_object**)realloc(
516 store->certificates,
517 (size_t)store->cert_capacity * sizeof(struct json_object*));
518 }
519 for (i = 0; i < cert_count; i++)
520 store->certificates[i] = certs[i];
521 store->cert_count = cert_count;
522 free(certs);
523 }
524 }
525
526 /* Revocations */
527 {
528 int rev_buf[MTC_MAX_CERTS];
529 int rev_count = mtc_db_load_revocations(store->db, rev_buf,
530 MTC_MAX_CERTS);
531 if (rev_count > 0) {
532 store->revocation_capacity = rev_count * 2;
533 store->revoked_indices = (int*)malloc(
534 (size_t)store->revocation_capacity * sizeof(int));
535 memcpy(store->revoked_indices, rev_buf,
536 (size_t)rev_count * sizeof(int));
537 store->revocation_count = rev_count;
538 }
539 }
540
541 printf("[store] loaded %d entries, %d certs, %d landmarks, "
542 "%d revocations from DB\n",
543 store->tree.size, store->cert_count, store->landmark_count,
544 store->revocation_count);
545 fflush(stdout);
546 return 0;
547 }
548
549 /* Load entries */
550 snprintf(path, sizeof(path), "%s/entries.json", store->data_dir);
551 sz = read_file(path, buf, (int)sizeof(buf) - 1);
552 if (sz > 0) {
553 buf[sz] = 0;
554 arr = json_tokener_parse(buf);
555 if (arr) {
556 count = (int)json_object_array_length(arr);
557 for (i = 0; i < count; i++) {
558 struct json_object *val;
559 entry = json_object_array_get_idx(arr, (size_t)i);
560 if (json_object_object_get_ex(entry, "hex", &val)) {
561 const char *hex = json_object_get_string(val);
562 int entry_sz = 0;
563 int j;
564
565 if (json_object_object_get_ex(entry, "size", &val))
566 entry_sz = json_object_get_int(val);
567
568 if (entry_sz > 0) {
569 uint8_t *entry_bytes = (uint8_t*)malloc(entry_sz);
570 for (j = 0; j < entry_sz && hex[j*2] && hex[j*2+1]; j++) {
571 unsigned int bv;
572 sscanf(hex + j * 2, "%02x", &bv);
573 entry_bytes[j] = (uint8_t)bv;
574 }
575 mtc_tree_append(&store->tree, entry_bytes, entry_sz);
576 free(entry_bytes);
577 }
578 }
579 }
580 json_object_put(arr);
581 }
582 }
583
584 /* Load certificates */
585 snprintf(path, sizeof(path), "%s/certificates.json", store->data_dir);
586 sz = read_file(path, buf, (int)sizeof(buf) - 1);
587 if (sz > 0) {
588 buf[sz] = 0;
589 arr = json_tokener_parse(buf);
590 if (arr) {
591 count = (int)json_object_array_length(arr);
592 for (i = 0; i < count; i++) {
593 struct json_object *cert = json_object_array_get_idx(arr, (size_t)i);
594 if (store->cert_count >= store->cert_capacity) {
595 store->cert_capacity *= 2;
596 store->certificates = (struct json_object**)realloc(
597 store->certificates,
598 (size_t)store->cert_capacity * sizeof(struct json_object*));
599 }
600 store->certificates[store->cert_count++] = json_object_get(cert);
601 }
602 json_object_put(arr);
603 }
604 }
605
606 /* Load landmarks */
607 snprintf(path, sizeof(path), "%s/landmarks.json", store->data_dir);
608 sz = read_file(path, buf, (int)sizeof(buf) - 1);
609 if (sz > 0) {
610 buf[sz] = 0;
611 arr = json_tokener_parse(buf);
612 if (arr) {
613 count = (int)json_object_array_length(arr);
614 for (i = 0; i < count && i < MTC_MAX_LANDMARKS; i++)
615 store->landmarks[store->landmark_count++] =
616 json_object_get_int(json_object_array_get_idx(arr, (size_t)i));
617 json_object_put(arr);
618 }
619 }
620
621 printf("[store] loaded %d entries, %d certs, %d landmarks\n",
622 store->tree.size, store->cert_count, store->landmark_count);
623 fflush(stdout);
624 return 0;
625}

631/******************************************************************************
632 * Function: mtc_store_add_entry
633 *
634 * Description:
635 * Appends a serialised entry to the Merkle tree and persists it to DB
636 * (if connected). Automatically records a landmark if the new tree
637 * size is a multiple of MTC_LANDMARK_INTERVAL.
638 *
639 * Input Arguments:
640 * store - Target store.
641 * entry - Serialised entry bytes (0x01 prefix = TBS, 0x00 = null).
642 * entrySz - Size in bytes.
643 *
644 * Returns:
645 * 0-based log index of the new entry.
646 *
647 * Side Effects:
648 * Appends to tree, may write to DB, may add a landmark.
649 ******************************************************************************/
650int mtc_store_add_entry(MtcStore *store, const uint8_t *entry, int entrySz)
651{
652 int idx = mtc_tree_append(&store->tree, entry, entrySz);
653
654 /* Persist entry to DB */
655 if (store->use_db && store->db) {
656 uint8_t lh[MTC_HASH_SIZE];
657 int entry_type = (entrySz > 0 && entry[0] == 0x01) ? 1 : 0;
658 const char *tbs_json = NULL;
659 char *tbs_str = NULL;
660
661 mtc_hash_leaf(entry, entrySz, lh);
662
663 if (entry_type == 1 && entrySz > 1) {
664 tbs_str = (char*)malloc((size_t)entrySz);
665 memcpy(tbs_str, entry + 1, (size_t)(entrySz - 1));
666 tbs_str[entrySz - 1] = 0;
667 tbs_json = tbs_str;
668 }
669
670 mtc_db_save_entry(store->db, idx, entry_type, tbs_json,
671 entry, entrySz, lh);
672 free(tbs_str);
673 }
674
675 /* Check for landmark */
676 if (store->tree.size % MTC_LANDMARK_INTERVAL == 0 &&
677 store->landmark_count < MTC_MAX_LANDMARKS) {
678 store->landmarks[store->landmark_count++] = store->tree.size;
679 if (store->use_db && store->db)
680 mtc_db_save_landmark(store->db, store->tree.size);
681 }
682
683 return idx;
684}

686/******************************************************************************
687 * Function: mtc_store_checkpoint
688 *
689 * Description:
690 * Creates a checkpoint for the current tree state: computes the root
691 * hash, builds a JSON object with log_id/tree_size/root_hash/timestamp,
692 * stores it in the checkpoints array, and persists to DB.
693 *
694 * Input Arguments:
695 * store - Store.
696 *
697 * Returns:
698 * New json_object checkpoint. Caller owns the reference and must
699 * call json_object_put() when done. The store also keeps a ref
700 * in its checkpoints array.
701 *
702 * Side Effects:
703 * Appends to store->checkpoints (up to 256). Writes to DB.
704 ******************************************************************************/
705struct json_object *mtc_store_checkpoint(MtcStore *store)
706{
707 uint8_t root[MTC_HASH_SIZE];
708 char root_hex[MTC_HASH_SIZE * 2 + 1];
709 struct json_object *cp;
710 int i;
711
712 mtc_tree_root_hash(&store->tree, store->tree.size, root);
713 for (i = 0; i < MTC_HASH_SIZE; i++)
714 snprintf(root_hex + i * 2, 3, "%02x", root[i]);
715
716 cp = json_object_new_object();
717 json_object_object_add(cp, "log_id",
718 json_object_new_string(store->log_id));
719 json_object_object_add(cp, "tree_size",
720 json_object_new_int(store->tree.size));
721 json_object_object_add(cp, "root_hash",
722 json_object_new_string(root_hex));
723 json_object_object_add(cp, "timestamp",
724 json_object_new_double(time(NULL)));
725
726 if (store->checkpoint_count < 256) {
727 store->checkpoints[store->checkpoint_count++] = json_object_get(cp);
728 }
729
730 /* Persist to DB */
731 if (store->use_db && store->db) {
732 mtc_db_save_checkpoint(store->db, store->log_id,
733 store->tree.size, root_hex, (double)time(NULL));
734 }
735
736 return cp;
737}

739/******************************************************************************
740 * Function: mtc_store_cosign
741 *
742 * Description:
743 * Cosigns a subtree range [start, end) with the CA's Ed25519 key.
744 * Builds the signature input per the MTC draft specification:
745 * "mtc-subtree/v1\n\0" + cosigner_id + log_id + start(8BE) +
746 * end(8BE) + subtree_hash
747 *
748 * Input Arguments:
749 * store - Store (provides CA key and tree).
750 * start - Subtree start (inclusive).
751 * end - Subtree end (exclusive).
752 * sig_out - Caller-owned buffer (>= 64 bytes).
753 * sig_sz - Receives the signature size.
754 *
755 * Returns:
756 * 0 on success, non-zero wolfCrypt error code on failure.
757 ******************************************************************************/
758int mtc_store_cosign(MtcStore *store, int start, int end,
759 uint8_t *sig_out, int *sig_sz)
760{
761 ed25519_key key;
762 uint8_t subtree_hash[MTC_HASH_SIZE];
763 uint8_t msg[256];
764 int msg_sz = 0;
765 word32 idx_w = 0;
766 word32 outSz;
767 int ret, i;
768
769 mtc_tree_subtree_hash(&store->tree, start, end, subtree_hash);
770
771 /* Build signature input per MTC draft specification:
772 * 16-byte context prefix (includes NUL) + cosigner_id + log_id +
773 * start as 8-byte big-endian + end as 8-byte big-endian + subtree hash */
774 memcpy(msg, "mtc-subtree/v1\n\x00", 16);
775 msg_sz = 16;
776 memcpy(msg + msg_sz, store->cosigner_id, strlen(store->cosigner_id));
777 msg_sz += (int)strlen(store->cosigner_id);
778 memcpy(msg + msg_sz, store->log_id, strlen(store->log_id));
779 msg_sz += (int)strlen(store->log_id);
780
781 for (i = 7; i >= 0; i--)
782 msg[msg_sz++] = (uint8_t)((start >> (i * 8)) & 0xff);
783 for (i = 7; i >= 0; i--)
784 msg[msg_sz++] = (uint8_t)((end >> (i * 8)) & 0xff);
785
786 memcpy(msg + msg_sz, subtree_hash, MTC_HASH_SIZE);
787 msg_sz += MTC_HASH_SIZE;
788
789 /* Sign with Ed25519 */
790 ret = wc_ed25519_init(&key);
791 if (ret != 0) return ret;
792
793 ret = wc_Ed25519PrivateKeyDecode(store->ca_priv_key, &idx_w, &key,
794 (word32)store->ca_priv_key_sz);
795 if (ret != 0) { wc_ed25519_free(&key); return ret; }
796
797 outSz = ED25519_SIG_SIZE;
798 ret = wc_ed25519_sign_msg(msg, (word32)msg_sz, sig_out, &outSz, &key);
799 *sig_sz = (int)outSz;
800
801 wc_ed25519_free(&key);
802 return ret;
803}

805/******************************************************************************
806 * Function: mtc_store_get_public_key_pem
807 *
808 * Description:
809 * Exports the CA Ed25519 public key as a PEM string by decoding the
810 * stored private key DER, extracting the public key DER, and
811 * converting to PEM.
812 *
813 * Input Arguments:
814 * store - Store (provides CA key).
815 * out - Caller-owned buffer for the PEM string.
816 * maxSz - Size of out in bytes.
817 *
818 * Returns:
819 * Number of PEM bytes written (> 0) on success.
820 * Negative wolfCrypt error code on failure.
821 ******************************************************************************/
822int mtc_store_get_public_key_pem(MtcStore *store, char *out, int maxSz)
823{
824 uint8_t der[128];
825 int derSz;
826 ed25519_key key;
827 word32 idx = 0;
828 int ret;
829
830 ret = wc_ed25519_init(&key);
831 if (ret != 0) return ret;
832
833 ret = wc_Ed25519PrivateKeyDecode(store->ca_priv_key, &idx, &key,
834 (word32)store->ca_priv_key_sz);
835 if (ret != 0) { wc_ed25519_free(&key); return ret; }
836
837 derSz = wc_Ed25519PublicKeyToDer(&key, der, sizeof(der), 1);
838 wc_ed25519_free(&key);
839
840 if (derSz < 0) return derSz;
841
842 ret = wc_DerToPem(der, (word32)derSz, (byte*)out, (word32)maxSz,
843 ED25519_TYPE);
844 return ret;
845}

851/******************************************************************************
852 * Function: mtc_store_revoke
853 *
854 * Description:
855 * Revokes a certificate by log index. The index is inserted into the
856 * sorted revoked_indices array (growing it if needed). The revocation
857 * is persisted to both DB and a local revocations.json file.
858 * If already revoked, returns 0 immediately (idempotent).
859 *
860 * Input Arguments:
861 * store - Store.
862 * cert_index - Log index of the certificate to revoke.
863 * reason - Human-readable reason (may be NULL).
864 *
865 * Returns:
866 * 0 on success (including already-revoked).
867 * -1 on memory allocation failure.
868 *
869 * Side Effects:
870 * Inserts into revoked_indices (sorted), writes to DB and file.
871 ******************************************************************************/
872int mtc_store_revoke(MtcStore *store, int cert_index, const char *reason)
873{
874 /* Check not already revoked */
875 if (mtc_store_is_revoked(store, cert_index))
876 return 0; /* Already revoked */
877
878 /* Grow array if needed */
879 if (store->revocation_count >= store->revocation_capacity) {
880 int newcap = store->revocation_capacity == 0 ? 64
881 : store->revocation_capacity * 2;
882 int *tmp = (int*)realloc(store->revoked_indices,
883 (size_t)newcap * sizeof(int));
884 if (!tmp) return -1;
885 store->revoked_indices = tmp;
886 store->revocation_capacity = newcap;
887 }
888
889 /* Insert into sorted position via insertion sort — maintains the
890 * invariant that revoked_indices is always sorted ascending for
891 * binary search in mtc_store_is_revoked(). */
892 {
893 int i = store->revocation_count;
894 while (i > 0 && store->revoked_indices[i - 1] > cert_index) {
895 store->revoked_indices[i] = store->revoked_indices[i - 1];
896 i--;
897 }
898 store->revoked_indices[i] = cert_index;
899 store->revocation_count++;
900 }
901
902 /* Persist to DB */
903 if (store->use_db && store->db)
904 mtc_db_save_revocation(store->db, cert_index, reason);
905
906 /* Persist to file */
907 {
908 char path[1024];
909 struct json_object *arr = json_object_new_array();
910 int i;
911 const char *s;
912 for (i = 0; i < store->revocation_count; i++)
913 json_object_array_add(arr,
914 json_object_new_int(store->revoked_indices[i]));
915 snprintf(path, sizeof(path), "%s/revocations.json", store->data_dir);
916 s = json_object_to_json_string(arr);
917 {
918 FILE *f = fopen(path, "w");
919 if (f) { fputs(s, f); fclose(f); }
920 }
921 json_object_put(arr);
922 }
923
924 printf("[store] revoked cert index %d (reason: %s)\n",
925 cert_index, reason ? reason : "unspecified");
926 return 0;
927}

929/******************************************************************************
930 * Function: mtc_store_is_revoked
931 *
932 * Description:
933 * Checks whether a certificate is revoked via binary search on the
934 * sorted revoked_indices array.
935 *
936 * Input Arguments:
937 * store - Store.
938 * cert_index - Log index to check.
939 *
940 * Returns:
941 * 1 if revoked.
942 * 0 if not revoked.
943 ******************************************************************************/
944int mtc_store_is_revoked(MtcStore *store, int cert_index)
945{
946 int lo = 0, hi = store->revocation_count - 1;
947 while (lo <= hi) {
948 int mid = (lo + hi) / 2;
949 if (store->revoked_indices[mid] == cert_index)
950 return 1;
951 else if (store->revoked_indices[mid] < cert_index)
952 lo = mid + 1;
953 else
954 hi = mid - 1;
955 }
956 return 0;
957}

959/******************************************************************************
960 * Function: mtc_store_get_revocation_list
961 *
962 * Description:
963 * Builds a signed revocation list as a JSON object containing:
964 * log_id, revoked (array of cert indices), count, updated_at
965 * timestamp, and an Ed25519 signature over the revoked array JSON.
966 *
967 * Input Arguments:
968 * store - Store.
969 *
970 * Returns:
971 * New json_object. Caller owns and must free with json_object_put().
972 ******************************************************************************/
973struct json_object *mtc_store_get_revocation_list(MtcStore *store)
974{
975 struct json_object *obj = json_object_new_object();
976 struct json_object *arr = json_object_new_array();
977 int i;
978 uint8_t sig[64];
979 int sig_sz = 0;
980
981 json_object_object_add(obj, "log_id",
982 json_object_new_string(store->log_id));
983
984 for (i = 0; i < store->revocation_count; i++)
985 json_object_array_add(arr,
986 json_object_new_int(store->revoked_indices[i]));
987 json_object_object_add(obj, "revoked", arr);
988
989 json_object_object_add(obj, "count",
990 json_object_new_int(store->revocation_count));
991 json_object_object_add(obj, "updated_at",
992 json_object_new_double((double)time(NULL)));
993
994 /* Sign the revocation list with the CA key */
995 {
996 const char *payload = json_object_to_json_string(arr);
997 ed25519_key key;
998 word32 idx = 0;
999 word32 outSz = sizeof(sig);
1000
1001 if (wc_ed25519_init(&key) == 0 &&
1002 wc_Ed25519PrivateKeyDecode(store->ca_priv_key, &idx, &key,
1003 (word32)store->ca_priv_key_sz) == 0) {
1004 if (wc_ed25519_sign_msg((const byte*)payload,
1005 (word32)strlen(payload), sig, &outSz, &key) == 0) {
1006 char sig_hex[129];
1007 sig_sz = (int)outSz;
1008 for (i = 0; i < sig_sz; i++)
1009 snprintf(sig_hex + i * 2, 3, "%02x", sig[i]);
1010 json_object_object_add(obj, "signature",
1011 json_object_new_string(sig_hex));
1012 }
1013 wc_ed25519_free(&key);
1014 }
1015 }
1016
1017 return obj;
1018}
