416/******************************************************************************
417 * Function: validate_ca_dns_txt
418 *
419 * Description:
420 * Queries DNS for _mtc-ca.<domain> TXT records and validates against
421 * the expected fingerprint (and optionally a nonce).
422 *
423 * Two TXT record formats are supported:
424 * v=mtc-ca1; fp=sha256:<hex> — legacy (fp-only)
425 * v=mtc-ca2; fp=sha256:<hex>; n=<nonce> — nonce-bound
426 *
427 * Field matching is exact (split on ';', trim whitespace) to prevent
428 * crafted records from bypassing validation.
429 *
430 * Input Arguments:
431 * domain - Domain name (e.g. "example.com"). The query is
432 * for _mtc-ca.<domain>.
433 * fp_hex - Expected SHA-256 fingerprint (64 hex chars).
434 * expected_nonce - If non-NULL, require v=mtc-ca2 with matching nonce.
435 * If NULL, accept legacy v=mtc-ca1 (fp-only).
436 *
437 * Returns:
438 * 1 if a matching TXT record is found.
439 * 0 if no match, DNS query failed, or parse error.
440 ******************************************************************************/
441static int validate_ca_dns_txt(const char *domain, const char *fp_hex,
442 const char *expected_nonce)
443{
444 char qname[256];
445 unsigned char answer[4096];
446 int ans_len, i;
447 ns_msg msg;
448 ns_rr rr;
449
450 snprintf(qname, sizeof(qname), "_mtc-ca.%s", domain);
451
452 ans_len = res_query(qname, ns_c_in, ns_t_txt, answer, sizeof(answer));
453 if (ans_len < 0) {
454 LOG_WARN("DNS query failed for %s", qname);
455 return 0;
456 }
457
458 if (ns_initparse(answer, ans_len, &msg) < 0) {
459 LOG_WARN("failed to parse DNS response for %s", qname);
460 return 0;
461 }
462
463 for (i = 0; i < ns_msg_count(msg, ns_s_an); i++) {
464 if (ns_parserr(&msg, ns_s_an, i, &rr) == 0 &&
465 ns_rr_type(rr) == ns_t_txt) {
466 const unsigned char *rdata = ns_rr_rdata(rr);
467 int rdlen = ns_rr_rdlen(rr);
468 if (rdlen > 1) {
469 int txt_len = rdata[0];
470 if (txt_len <= rdlen - 1) {
471 char txt[512];
472 if (txt_len >= (int)sizeof(txt))
473 txt_len = (int)sizeof(txt) - 1;
474 memcpy(txt, rdata + 1, txt_len);
475 txt[txt_len] = '\0';
476 LOG_TRACE("TXT record: \"%s\"", txt);
477
478 /* Parse TXT record fields by splitting on ';'
479 * and matching exact key=value pairs. No substring
480 * matching — prevents crafted records. */
481 {
482 char tmp[512];
483 char *field, *saveptr;
484 const char *v_val = NULL, *fp_val = NULL;
485 const char *n_val = NULL;
486
487 snprintf(tmp, sizeof(tmp), "%s", txt);
488 for (field = strtok_r(tmp, ";", &saveptr);
489 field != NULL;
490 field = strtok_r(NULL, ";", &saveptr)) {
491 /* Skip leading whitespace */
492 while (*field == ' ') field++;
493 if (strncmp(field, "v=", 2) == 0)
494 v_val = field + 2;
495 else if (strncmp(field, "fp=sha256:", 10) == 0)
496 fp_val = field + 10;
497 else if (strncmp(field, "n=", 2) == 0)
498 n_val = field + 2;
499 }
500
501 if (expected_nonce) {
502 /* v=mtc-ca2: exact field matching */
503 if (v_val && strcmp(v_val, "mtc-ca2") == 0 &&
504 fp_val && strcmp(fp_val, fp_hex) == 0 &&
505 n_val && strcmp(n_val, expected_nonce) == 0) {
506 LOG_DEBUG("v=mtc-ca2 MATCH for %s", qname);
507 return 1;
508 }
509 }
510 else {
511 /* Legacy v=mtc-ca1: fingerprint-only */
512 if (v_val && strcmp(v_val, "mtc-ca1") == 0 &&
513 fp_val && strcmp(fp_val, fp_hex) == 0) {
514 LOG_DEBUG("v=mtc-ca1 MATCH for %s", qname);
515 return 1;
516 }
517 }
518 }
519 }
520 }
521 }
522 }
523
524 LOG_WARN("no matching TXT record for %s", qname);
525 return 0;
526}

528/******************************************************************************
529 * Function: validate_ca_cert_if_present
530 *
531 * Description:
532 * If the request extensions contain a ca_certificate_pem, parses the
533 * X.509 certificate, verifies CA:TRUE in Basic Constraints, extracts
534 * the SAN DNS name and SPKI SHA-256 fingerprint, and validates domain
535 * ownership via DNS TXT record.
536 *
537 * Root CAs (pathlen absent or > 0) skip DNS validation — only
538 * intermediate CAs (pathlen == 0) require a _mtc-ca.<domain> record.
539 *
540 * If no ca_certificate_pem is present, the request is not a CA
541 * enrollment and validation is trivially passed.
542 *
543 * Input Arguments:
544 * extensions - Request extensions json_object (may be NULL).
545 * enrollment_nonce - Nonce for v=mtc-ca2 validation (NULL = legacy).
546 *
547 * Returns:
548 * 1 if not a CA request, or CA validated successfully.
549 * 0 if CA validation failed (rejected).
550 ******************************************************************************/
551static int validate_ca_cert_if_present(struct json_object *extensions,
552 const char *enrollment_nonce)
553{
554 struct json_object *ca_cert_val;
555 const char *ca_cert_pem;
556 DecodedCert decoded;
557 int ret;
558 const unsigned char *pem_bytes;
559 unsigned char der_buf[8192];
560 int der_sz;
561 int pem_len;
562 char fp_hex[65];
563
564 if (!extensions)
565 return 1; /* No extensions, not a CA request */
566
567 if (!json_object_object_get_ex(extensions, "ca_certificate_pem", &ca_cert_val))
568 return 1; /* No CA cert in request, OK */
569
570 ca_cert_pem = json_object_get_string(ca_cert_val);
571 if (!ca_cert_pem || strlen(ca_cert_pem) == 0)
572 return 1;
573
574 /* Root CA detection is server-side only — never trust the client's
575 * "root_ca" flag. We determine root status from the parsed certificate's
576 * Basic Constraints: pathlen absent or > 0 = root, pathlen == 0 =
577 * intermediate. The check happens below after parsing the cert. */
578
579 LOG_DEBUG("CA certificate PEM found, validating...");
580
581 /* Convert PEM to DER */
582 pem_bytes = (const unsigned char *)ca_cert_pem;
583 pem_len = (int)strlen(ca_cert_pem);
584 LOG_TRACE("PEM length: %d bytes", pem_len);
585 if (pem_len > 6000) {
586 LOG_WARN("CA cert PEM too large");
587 return 0;
588 }
589 der_sz = (int)sizeof(der_buf);
590 ret = wc_CertPemToDer(pem_bytes, pem_len, der_buf, der_sz, CERT_TYPE);
591 if (ret < 0) {
592 LOG_WARN("PEM to DER conversion failed: %d", ret);
593 return 0;
594 }
595 der_sz = ret;
596 LOG_TRACE("DER size: %d bytes", der_sz);
597
598 /* Parse the certificate */
599 wc_InitDecodedCert(&decoded, der_buf, (word32)der_sz, NULL);
600 ret = wc_ParseCert(&decoded, CERT_TYPE, NO_VERIFY, NULL);
601 if (ret != 0) {
602 LOG_WARN("certificate parse failed: %d", ret);
603 wc_FreeDecodedCert(&decoded);
604 return 0;
605 }
606
607 /* Check Basic Constraints: CA:TRUE */
608 if (!decoded.isCA) {
609 LOG_WARN("certificate is not a CA (isCA=0)");
610 wc_FreeDecodedCert(&decoded);
611 return 0;
612 }
613 LOG_DEBUG("CA:TRUE, pathlen:%d", decoded.pathLength);
614
615 /* Server-side root CA detection: pathlen absent (pathLengthSet==0) or
616 * pathlen > 0 means root. pathlen == 0 means intermediate — requires
617 * DNS validation. */
618 if (!decoded.pathLengthSet || decoded.pathLength > 0) {
619 LOG_INFO("root CA detected (pathLengthSet=%d, pathlen=%u) — "
620 "DNS validation skipped",
621 decoded.pathLengthSet, decoded.pathLength);
622 wc_FreeDecodedCert(&decoded);
623 return 1;
624 }
625
626 /* Extract SAN DNS name */
627 {
628 DNS_entry *san = decoded.altNames;
629 char domain[256] = {0};
630
631 while (san) {
632 if (san->type == ASN_DNS_TYPE && san->name) {
633 snprintf(domain, sizeof(domain), "%s", san->name);
634 break;
635 }
636 san = san->next;
637 }
638
639 if (domain[0] == '\0') {
640 LOG_WARN("no SAN DNS name found in CA cert");
641 wc_FreeDecodedCert(&decoded);
642 return 0;
643 }
644
645 LOG_DEBUG("SAN DNS: %s", domain);
646
647 /* Compute SHA-256 fingerprint of SubjectPublicKeyInfo DER */
648 {
649 wc_Sha256 sha;
650 uint8_t h[32];
651 uint8_t spki_buf[1024];
652 word32 spki_sz = sizeof(spki_buf);
653
654 ret = wc_GetSubjectPubKeyInfoDerFromCert(
655 der_buf, (word32)der_sz, spki_buf, &spki_sz);
656 if (ret != 0) {
657 LOG_WARN("failed to extract SPKI: %d", ret);
658 wc_FreeDecodedCert(&decoded);
659 return 0;
660 }
661
662 wc_InitSha256(&sha);
663 wc_Sha256Update(&sha, spki_buf, spki_sz);
664 wc_Sha256Final(&sha, h);
665 wc_Sha256Free(&sha);
666 to_hex(h, 32, fp_hex);
667 }
668
669 LOG_DEBUG("public key fingerprint: %.16s...", fp_hex);
670
671 wc_FreeDecodedCert(&decoded);
672
673 /* Check DNS — pass nonce for v=mtc-ca2, or NULL for legacy */
674 return validate_ca_dns_txt(domain, fp_hex, enrollment_nonce);
675 }

678/******************************************************************************
679 * Function: handle_enrollment_nonce
680 *
681 * Description:
682 * POST /enrollment/nonce — issue a server-side enrollment nonce.
683 *
684 * CA enrollment (type=ca or omitted):
685 * Body: {"domain": "...", "public_key_fingerprint": "sha256:..."}
686 * No CA needs to exist — DNS TXT validates domain ownership.
687 *
688 * Leaf enrollment (type=leaf):
689 * Body: {"domain": "...", "public_key_fingerprint": "sha256:...",
690 * "type": "leaf"}
691 * A registered CA must exist for this domain.
692 *
693 * Returns a JSON response with the nonce, expiration, and (for CA
694 * nonces) the DNS record the caller must create.
695 *
696 * Input Arguments:
697 * io - Client I/O context.
698 * store - MTC store (DB connection used for nonce storage).
699 * body - HTTP request body (JSON).
700 * body_len - Length of body in bytes.
701 *
702 * Side Effects:
703 * Creates a pending nonce row in mtc_enrollment_nonces.
704 ******************************************************************************/
705static void handle_enrollment_nonce(client_io *io, MtcStore *store,
706 const char *body, int body_len)
707{
708 struct json_object *req, *val;
709 const char *domain, *fp_raw, *nonce_type;
710 char fp_hex[65];
711 char nonce[MTC_NONCE_HEX_LEN + 1];
712 long expires;
713 struct json_object *resp;
714 int ret, ca_index = -1;
715 int is_leaf = 0;
716
717 (void)body_len;
718
719 req = json_tokener_parse(body);
720 if (!req) {
721 http_send_error(io, 400, "invalid JSON");
722 return;
723 }
724
725 if (!json_object_object_get_ex(req, "domain", &val)) {
726 http_send_error(io, 400, "missing 'domain'");
727 json_object_put(req);
728 return;
729 }
730 domain = json_object_get_string(val);
731
732 if (!json_object_object_get_ex(req, "public_key_fingerprint", &val)) {
733 http_send_error(io, 400, "missing 'public_key_fingerprint'");
734 json_object_put(req);
735 return;
736 }
737 fp_raw = json_object_get_string(val);
738
739 /* Strip "sha256:" prefix and validate hex format */
740 if (strncmp(fp_raw, "sha256:", 7) == 0)
741 fp_raw += 7;
742 if (strlen(fp_raw) != 64) {
743 http_send_error(io, 400, "fingerprint must be exactly 64 hex chars");
744 json_object_put(req);
745 return;
746 }
747 {
748 int fi;
749 for (fi = 0; fi < 64; fi++) {
750 char c = fp_raw[fi];
751 if (!((c >= '0' && c <= '9') || (c >= 'a' && c <= 'f') ||
752 (c >= 'A' && c <= 'F'))) {
753 http_send_error(io, 400, "fingerprint contains non-hex chars");
754 json_object_put(req);
755 return;
756 }
757 }
758 }
759 snprintf(fp_hex, sizeof(fp_hex), "%s", fp_raw);
760
761 /* Check nonce type */
762 nonce_type = "ca";
763 if (json_object_object_get_ex(req, "type", &val))
764 nonce_type = json_object_get_string(val);
765 is_leaf = (strcmp(nonce_type, "leaf") == 0);
766
767 /* Rate limit: leaf nonces (10/min, 100/hr) vs CA nonces (3/min, 10/hr) */
768 if (!mtc_ratelimit_check(io->ip_str, is_leaf ? RL_NONCE_LEAF : RL_NONCE_CA)) {
769 http_send_error(io, 429, "rate limit exceeded");
770 json_object_put(req);
771 return;
772 }
773
774 /* For leaf nonces, verify a registered CA exists for this domain */
775 if (is_leaf) {
776 ca_index = mtc_db_find_ca_for_domain(store->db, domain);
777 if (ca_index < 0) {
778 http_send_error(io, 403,
779 "no registered CA exists for this domain — "
780 "enroll a CA first via enroll-ca");
781 json_object_put(req);
782 return;
783 }
784 LOG_INFO("leaf nonce requested for %s (authorized by CA index %d)",
785 domain, ca_index);
786 }
787
788 /* Create nonce in DB */
789 ret = mtc_db_create_nonce(store->db, domain, fp_hex, ca_index,
790 nonce, &expires);
791 if (ret == -1) {
792 http_send_error(io, 409,
793 "pending enrollment already exists for this domain and key");
794 json_object_put(req);
795 return;
796 }
797 if (ret < 0) {
798 http_send_error(io, 500, "nonce generation failed");
799 json_object_put(req);
800 return;
801 }
802
803 LOG_INFO("%s nonce issued for %s (fp=%.16s..., expires=%ld)",
804 is_leaf ? "leaf" : "CA", domain, fp_hex, expires);
805
806 resp = json_object_new_object();
807 json_object_object_add(resp, "nonce", json_object_new_string(nonce));
808 json_object_object_add(resp, "expires", json_object_new_int64(expires));
809 json_object_object_add(resp, "type",
810 json_object_new_string(is_leaf ? "leaf" : "ca"));
811 if (ca_index >= 0)
812 json_object_object_add(resp, "ca_index",
813 json_object_new_int(ca_index));
814
815 if (!is_leaf) {
816 /* CA nonce: include DNS record to create */
817 char dns_name[256], dns_value[512];
818 snprintf(dns_name, sizeof(dns_name), "_mtc-ca.%s.", domain);
819 snprintf(dns_value, sizeof(dns_value),
820 "v=mtc-ca2; fp=sha256:%s; n=%s; exp=%ld",
821 fp_hex, nonce, expires);
822 json_object_object_add(resp, "dns_record_name",
823 json_object_new_string(dns_name));
824 json_object_object_add(resp, "dns_record_value",
825 json_object_new_string(dns_value));
826 }
827
828 http_send_json_obj(io, 200, resp);
829 json_object_put(resp);
830 json_object_put(req);
831}

833/******************************************************************************
834 * Function: handle_certificate_request
835 *
836 * Description:
837 * POST /certificate/request — enroll a CA or leaf certificate.
838 *
839 * For CA enrollment: validates via DNS TXT record at _mtc-ca.<domain>.
840 * For leaf enrollment: requires and atomically consumes a valid nonce.
841 * Both paths are gated by an enrollment-level AbuseIPDB check
842 * (ABUSEIPDB_ENROLL_THRESHOLD, stricter than general access).
843 *
844 * On success: adds a TBS entry to the Merkle tree, generates an
845 * inclusion proof, cosigns, persists to disk and DB, and returns
846 * a standalone certificate (HTTP 201).
847 *
848 * Input Arguments:
849 * io - Client I/O context.
850 * store - MTC store (tree, certs, DB).
851 * body - HTTP request body (JSON).
852 * body_len - Length of body in bytes.
853 *
854 * Side Effects:
855 * Appends to the Merkle tree, creates a checkpoint, persists to
856 * disk and DB, grows the certificate array if needed.
857 ******************************************************************************/
858static void handle_certificate_request(client_io *io, MtcStore *store,
859 const char *body, int body_len)
860{
861 struct json_object *req, *val;
862 const char *subject, *pub_key_pem, *key_algo;
863 (void)body_len;
864 int validity_days;
865 struct json_object *extensions = NULL;
866
867 /* Enrollment-level AbuseIPDB gate (stricter than general access) */
868 if (io->ip_str[0] != '\0') {
869 int score = mtc_checkendpoint(io->ip_str);
870 if (score >= ABUSEIPDB_ENROLL_THRESHOLD) {
871 LOG_WARN("enrollment rejected for %s (abuse score %d >= %d)",
872 io->ip_str, score, ABUSEIPDB_ENROLL_THRESHOLD);
873 http_send_error(io, 403, "enrollment denied");
874 return;
875 }
876 }
877
878 /* Parse request */
879 req = json_tokener_parse(body);
880 if (!req) {
881 http_send_error(io, 400, "invalid JSON");
882 return;
883 }
884
885 if (!json_object_object_get_ex(req, "subject", &val)) {
886 http_send_error(io, 400, "missing 'subject'");
887 json_object_put(req);
888 return;
889 }
890 subject = json_object_get_string(val);
891
892 if (!json_object_object_get_ex(req, "public_key_pem", &val)) {
893 http_send_error(io, 400, "missing 'public_key_pem'");
894 json_object_put(req);
895 return;
896 }
897 pub_key_pem = json_object_get_string(val);
898
899 key_algo = "EC-P256";
900 if (json_object_object_get_ex(req, "key_algorithm", &val)) {
901 const char *requested = json_object_get_string(val);
902 if (strcmp(requested, "EC-P256") != 0 &&
903 strcmp(requested, "EC-P384") != 0 &&
904 strcmp(requested, "Ed25519") != 0 &&
905 strcmp(requested, "ML-DSA-44") != 0 &&
906 strcmp(requested, "ML-DSA-65") != 0 &&
907 strcmp(requested, "ML-DSA-87") != 0) {
908 http_send_error(io, 400, "unsupported key_algorithm");
909 json_object_put(req);
910 return;
911 }
912 key_algo = requested;
913 }
914
915 validity_days = 90;
916 if (json_object_object_get_ex(req, "validity_days", &val)) {
917 validity_days = json_object_get_int(val);
918 if (validity_days < 1 || validity_days > 3650) {
919 http_send_error(io, 400,
920 "validity_days must be between 1 and 3650");
921 json_object_put(req);
922 return;
923 }
924 }
925
926 json_object_object_get_ex(req, "extensions", &extensions);
927
928 /* Enrollment authorization */
929 {
930 const char *enrollment_nonce = NULL;
931 int is_ca_enrollment = 0;
932
933 if (json_object_object_get_ex(req, "enrollment_nonce", &val))
934 enrollment_nonce = json_object_get_string(val);
935
936 /* Check if this is a CA enrollment (has ca_certificate_pem) */
937 if (extensions) {
938 struct json_object *ca_val;
939 if (json_object_object_get_ex(extensions, "ca_certificate_pem",
940 &ca_val))
941 is_ca_enrollment = 1;
942 }
943
944 if (is_ca_enrollment) {
945 /* CA enrollment: DNS TXT validation, no nonce required.
946 * The DNS record at _mtc-ca.<domain> proves domain ownership. */
947 if (!validate_ca_cert_if_present(extensions, NULL)) {
948 http_send_error(io, 403,
949 "CA certificate rejected: DNS validation failed — "
950 "no matching _mtc-ca.<domain> TXT record found");
951 json_object_put(req);
952 return;
953 }
954 }
955 else {
956 /* Leaf enrollment: nonce required.
957 * The CA operator issues the nonce (POST /enrollment/nonce
958 * with type=leaf), then sends it to the leaf out-of-band.
959 * The server verifies the nonce is valid, bound to this
960 * domain + key, and was authorized by a registered CA. */
961 if (!enrollment_nonce) {
962 http_send_error(io, 400,
963 "enrollment_nonce required for leaf enrollment — "
964 "obtain a nonce from your CA operator");
965 json_object_put(req);
966 return;
967 }
968
969 /* Compute leaf's SPKI fingerprint for nonce binding check.
970 * The nonce was bound to domain + fingerprint at creation.
971 * We must verify both match — not just the domain. */
972 {
973 wc_Sha256 sha;
974 uint8_t h[32];
975 char leaf_fp[65];
976 int fi;
977
978 wc_InitSha256(&sha);
979 wc_Sha256Update(&sha, (const uint8_t *)pub_key_pem,
980 (word32)strlen(pub_key_pem));
981 wc_Sha256Final(&sha, h);
982 wc_Sha256Free(&sha);
983 for (fi = 0; fi < 32; fi++)
984 snprintf(leaf_fp + fi * 2, 3, "%02x", h[fi]);
985
986 /* Atomic validate + consume — prevents TOCTOU race.
987 * Checks nonce + domain + fingerprint in one UPDATE. */
988 if (!store->db ||
989 !mtc_db_validate_and_consume_nonce(store->db,
990 enrollment_nonce, subject, leaf_fp)) {
991 http_send_error(io, 403,
992 "invalid, expired, or already-used enrollment nonce");
993 json_object_put(req);
994 return;
995 }
996
997 LOG_INFO("leaf enrollment for '%s' authorized by nonce %.16s...",
998 subject, enrollment_nonce);
999 }
1000 }
1001 }
1002
1003 /* Build TBS entry */
1004 {
1005 struct json_object *tbs, *sc, *result, *checkpoint;
1006 struct json_object *proof_arr, *cosig_arr, *cosig;
1007 uint8_t *entry_buf = NULL;
1008 int entry_sz;
1009 int index;
1010 double now = (double)time(NULL);
1011 char spk_hash[65];
1012 uint8_t *proof = NULL;
1013 int proof_count = 0;
1014 uint8_t subtree_hash[MTC_HASH_SIZE];
1015 char hash_hex[MTC_HASH_SIZE * 2 + 1];
1016 uint8_t sig[64];
1017 int sig_sz = 0;
1018 int i, start, end;
1019
1020 /* Hash the public key */
1021 {
1022 wc_Sha256 sha;
1023 uint8_t h[32];
1024 wc_InitSha256(&sha);
1025 wc_Sha256Update(&sha, (const byte*)pub_key_pem,
1026 (word32)strlen(pub_key_pem));
1027 wc_Sha256Final(&sha, h);
1028 wc_Sha256Free(&sha);
1029 to_hex(h, 32, spk_hash);
1030 }
1031
1032 /* Build TBS JSON for serialization */
1033 tbs = json_object_new_object();
1034 json_object_object_add(tbs, "subject",
1035 json_object_new_string(subject));
1036 json_object_object_add(tbs, "subject_public_key_algorithm",
1037 json_object_new_string(key_algo));
1038 json_object_object_add(tbs, "subject_public_key_hash",
1039 json_object_new_string(spk_hash));
1040 json_object_object_add(tbs, "not_before",
1041 json_object_new_double(now));
1042 json_object_object_add(tbs, "not_after",
1043 json_object_new_double(now + validity_days * 86400.0));
1044 json_object_object_add(tbs, "extensions",
1045 extensions ? json_object_get(extensions) : json_object_new_object());
1046
1047 /* Serialize for Merkle tree: 0x01 + deterministic JSON */
1048 {
1049 struct json_object *ser = json_object_new_object();
1050 const char *ser_str;
1051 json_object_object_add(ser, "extensions",
1052 json_object_get(json_object_object_get(tbs, "extensions")));
1053 json_object_object_add(ser, "not_after",
1054 json_object_new_double(now + validity_days * 86400.0));
1055 json_object_object_add(ser, "not_before",
1056 json_object_new_double(now));
1057 json_object_object_add(ser, "spk_algorithm",
1058 json_object_new_string(key_algo));
1059 json_object_object_add(ser, "spk_hash",
1060 json_object_new_string(spk_hash));
1061 json_object_object_add(ser, "subject",
1062 json_object_new_string(subject));
1063
1064 ser_str = json_object_to_json_string_ext(ser,
1065 JSON_C_TO_STRING_PLAIN);
1066 entry_sz = 1 + (int)strlen(ser_str);
1067 entry_buf = (uint8_t *)malloc(entry_sz);
1068 entry_buf[0] = 0x01;
1069 memcpy(entry_buf + 1, ser_str, strlen(ser_str));
1070 json_object_put(ser);
1071 }
1072
1073 /* Add to log */
1074 index = mtc_store_add_entry(store, entry_buf, entry_sz);
1075
1076 /* Checkpoint */
1077 checkpoint = mtc_store_checkpoint(store);
1078
1079 /* Proof */
1080 start = 0;
1081 end = store->tree.size;
1082 mtc_tree_inclusion_proof(&store->tree, index, start, end,
1083 &proof, &proof_count);
1084 mtc_tree_subtree_hash(&store->tree, start, end, subtree_hash);
1085
1086 /* Cosign */
1087 mtc_store_cosign(store, start, end, sig, &sig_sz);
1088
1089 /* Build standalone certificate */
1090 sc = json_object_new_object();
1091 json_object_object_add(sc, "index", json_object_new_int(index));
1092 json_object_object_add(sc, "tbs_entry", json_object_get(tbs));
1093
1094 proof_arr = json_object_new_array();
1095 for (i = 0; i < proof_count; i++) {
1096 to_hex(proof + i * MTC_HASH_SIZE, MTC_HASH_SIZE, hash_hex);
1097 json_object_array_add(proof_arr,
1098 json_object_new_string(hash_hex));
1099 }
1100 json_object_object_add(sc, "inclusion_proof", proof_arr);
1101
1102 json_object_object_add(sc, "subtree_start",
1103 json_object_new_int(start));
1104 json_object_object_add(sc, "subtree_end",
1105 json_object_new_int(end));
1106
1107 to_hex(subtree_hash, MTC_HASH_SIZE, hash_hex);
1108 json_object_object_add(sc, "subtree_hash",
1109 json_object_new_string(hash_hex));
1110
1111 /* Cosignature */
1112 cosig_arr = json_object_new_array();
1113 cosig = json_object_new_object();
1114 json_object_object_add(cosig, "cosigner_id",
1115 json_object_new_string(store->cosigner_id));
1116 json_object_object_add(cosig, "log_id",
1117 json_object_new_string(store->log_id));
1118 json_object_object_add(cosig, "start", json_object_new_int(start));
1119 json_object_object_add(cosig, "end", json_object_new_int(end));
1120 json_object_object_add(cosig, "subtree_hash",
1121 json_object_new_string(hash_hex));
1122 {
1123 char sig_hex[64 * 2 + 1];
1124 to_hex(sig, sig_sz, sig_hex);
1125 json_object_object_add(cosig, "signature",
1126 json_object_new_string(sig_hex));
1127 }
1128 json_object_object_add(cosig, "algorithm",
1129 json_object_new_string("Ed25519"));
1130 json_object_array_add(cosig_arr, cosig);
1131 json_object_object_add(sc, "cosignatures", cosig_arr);
1132 json_object_object_add(sc, "trust_anchor_id",
1133 json_object_new_string(store->log_id));
1134
1135 /* Build result */
1136 result = json_object_new_object();
1137 json_object_object_add(result, "index", json_object_new_int(index));
1138 json_object_object_add(result, "standalone_certificate", sc);
1139 json_object_object_add(result, "checkpoint",
1140 json_object_get(checkpoint));
1141
1142 /* Store certificate */
1143 if (index >= store->cert_capacity) {
1144 store->cert_capacity *= 2;
1145 store->certificates = (struct json_object**)realloc(
1146 store->certificates,
1147 (size_t)store->cert_capacity * sizeof(struct json_object*));
1148 }
1149 while (store->cert_count <= index) {
1150 store->certificates[store->cert_count++] = NULL;
1151 }
1152 store->certificates[index] = json_object_get(result);
1153
1154 /* Persist */
1155 mtc_store_save(store);
1156 if (store->use_db && store->db) {
1157 const char *cert_str = json_object_to_json_string(result);
1158 mtc_db_save_certificate(store->db, index, cert_str);
1159 }
1160
1161 http_send_json_obj(io, 201, result);
1162
1163 json_object_put(result);
1164 json_object_put(tbs);
1165 json_object_put(checkpoint);
1166 free(proof);
1167 free(entry_buf);
1168 }
1169
1170 json_object_put(req);
1171}

1235/* GET /log/consistency?old=N&new=M — Merkle consistency proof. */
1236static void handle_consistency(client_io *io, MtcStore *store, const char *path)
1237{
1238 /* Parse ?old=N&new=M from the query string */
1239 const char *qs;
1240 int old_size = 0, new_size = 0;
1241 uint8_t *proof = NULL;
1242 int proof_count = 0;
1243 uint8_t old_root[MTC_HASH_SIZE], new_root[MTC_HASH_SIZE];
1244 char hash_hex[MTC_HASH_SIZE * 2 + 1];
1245 struct json_object *obj, *proof_arr;
1246 int i;
1247
1248 qs = strchr(path, '?');
1249 if (qs) {
1250 const char *p = qs + 1;
1251 while (*p) {
1252 if (strncmp(p, "old=", 4) == 0)
1253 old_size = safe_atoi(p + 4, 10000000);
1254 else if (strncmp(p, "new=", 4) == 0)
1255 new_size = safe_atoi(p + 4, 10000000);
1256 p = strchr(p, '&');
1257 if (!p) break;
1258 p++;
1259 }
1260 }
1261
1262 if (old_size < 1 || new_size > store->tree.size || old_size > new_size) {
1263 http_send_error(io, 400, "invalid sizes");
1264 return;
1265 }
1266
1267 if (mtc_tree_consistency_proof(&store->tree, old_size, new_size,
1268 &proof, &proof_count) != 0) {

1539 * Description:
1540 * Reads a single HTTP request from the client, parses method and path,
1541 * extracts the body (reading additional data if Content-Length indicates

1546 * io - Client I/O context.
